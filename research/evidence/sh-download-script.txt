/*
 * 삭제된 파일 fileSeq 생성
 * 
 * controlId : 이노릭스 객체 ID
 * formId : Form 객체 ID
 * controlUse : uploadComplete 후 호출인지 아닌지
 * 
 * */
function makeInnoDelList(controlId, formId, controlUse){
	var tempDel = fn_nvl(delArr);	//기존 파일중 삭제된 파일 정보
	var strHtml;
	if(!fn_is_blank(tempDel)){
		delInfo = delArr;

		for(var i = 0; i < delInfo.length; i++){
			if(!controlUse){
				//strHtml = $('<input type="hidden" name="innorixDeletedFileName" value="'+delInfo[i].name+'">');
				strHtml = $('<input type="hidden" name="innorixDeletedFileName" value="'+delInfo[i].printFileName+'">');
				$("#" + formId).append(strHtml);
			}
    		
    		var sPoint = delInfo[i].downloadUrl.indexOf("fileSeq=") + 8;
    		var ePoint = delInfo[i].downloadUrl.indexOf("&_StartOffset");
    		
    		//IE일 경우 _StartOffset 파라미터가 안붙는다....
    		if(ePoint == -1){
    			ePoint = delInfo[i].downloadUrl.length;
    		}
    		
    		strHtml = $('<input type="hidden" name="innorixDeletedFileSeq" value="'+delInfo[i].downloadUrl.substring(sPoint,ePoint)+'">');
        	$("#" + formId).append(strHtml);
        }
	}
}

/*
 * 파일 확장자 체크 후 file_tp 데이터 리턴
 * 
 * name : 파일 명
 * 
 * */
function checkExt(name){

	if(name.indexOf("_Poster.") > 0){
		return "P";
	}
	if(name.indexOf("_Thumbnail.") > 0){
		return "T";
	}
	if(name.indexOf("_Movie.") > 0){
		return "M";
	}
	if(name.indexOf("_Brochure.") > 0){
		return "B";
	}
	if(name.indexOf("_jogam.") > 0){
		return "J";
	}
	if(name.indexOf("_location.") > 0){
		return "L";
	}
	if(name.indexOf("_batch.") > 0){
		return "BA";
	}
	return "A";
}

/*
 * 필수 파일 체크
 * 
 * brdTp : 게시판 형태
 * controlId : 이노릭스 객체 ID
 * brdTp : 2[이미지], 3[이미지+동영상(브로셔)]
 * 
 * */
function innoFileValidation(brdTp, brdId, type, controlId){
	
	if(!checkFileName(controlId)){
		alert("파일명에서 [ , ]를 제거해 주시기 바랍니다."); 
		return false;
	}
	
	if(brdTp == '2' || brdTp == '3'){
		
		var aList = control.getAllFiles();
		var dList = [];
		var uList = [];
		
		for(var i=0; i<aList.length; i++) {
			if(aList[i].transferType == "download") {
				dList.push(aList[i]);
			} else if(aList[i].transferType == "upload") {
				uList.push(aList[i]);
			}
		}
		
		console.log("##### innoFileValidation.dList : ", dList);
		console.log("##### innoFileValidation.uList : ", uList);
		
		var flag = 0;
		
		if(!fn_is_blank(dList)){
			for(var i = 0; i < dList.length; i++){
				if(checkExt(dList[i].printFileName) == "P"){
					flag = flag + 1;
				}
			}
		}
		
		if(!fn_is_blank(uList)){
			for(var i = 0; i < uList.length; i++){
				if(checkExt(uList[i].printFileName) == "P"){
					flag = flag + 1;
				}
			}
		}

		if(flag == 0){
			alert("이미지를 등록해주시기 바랍니다.\n[파일명 : \"_Poster.\" 포함]");
			return false;
		}
		
		if(flag > 1){
			alert("이미지 파일은 하나만 첨부가능합니다.");
			return false;
		}
		
		if(brdTp == '3' && type != 'U' && type != 'E'){
			//초기화
			flag = 0;
			
			if(!fn_is_blank(dList)){
				for(var i = 0; i < dList.length; i++){
					if(brdId == 'KS070801'){
						if(checkExt(dList[i].printFileName) == "M" || checkExt(dList[i].printFileName) == "B"){
							flag = flag + 1;
						}
					}else{
						if(checkExt(uList[i].printFileName) == "M"){
							flag = flag + 1;
						}
					}
				}
			}
			
			if(!fn_is_blank(uList)){
				for(var i = 0; i < uList.length; i++){
					if(brdId == 'KS070801'){
						if(checkExt(uList[i].printFileName) == "M" || checkExt(uList[i].printFileName) == "B"){
							flag = flag + 1;
						}
					}else{
						if(checkExt(uList[i].printFileName) == "M"){
							flag = flag + 1;
						}
					}
				}
			}
			
			var msg = '동영상';
			var caution = '\"_Movie.\"';
			
			if(brdId == 'KS070801'){
				if(type == 'B'){
					msg = '브로셔';
					caution = '\"_Brochure.\"';
				}
			}
			
			if(flag == 0){
				alert(msg + "파일을 등록해주시기 바랍니다.\n[파일명 : "+caution+" 포함]");
				return false;
			}
			
			if(flag > 1){
				alert(msg + "파일은 하나만 첨부가능합니다.");
				return false;
			}	
		}else if(brdTp == '3' && type == 'U'){
			flag = 0;
			
			if(!fn_is_blank(dList)){
				flag += dList.length;
			}
			
			if(!fn_is_blank(uList)){
				flag += uList.length;
			}
			
			if(flag > 1){
				alert("유투브 유형일 경우 이미지 파일만 첨부가능합니다.");
				return false;
			}
		}
		
		return true;
	} 
	
	return true;
}

//사용자 브라우저 정보 가져오기
function get_browser_info(){
    var ua=navigator.userAgent,tem,M=ua.match(/(opera|chrome|safari|firefox|msie|trident(?=\/))\/?\s*(\d+)/i) || []; 
    if(/trident/i.test(M[1])){
        tem=/\brv[ :]+(\d+)/g.exec(ua) || []; 
        return {name:'IE',version:(tem[1]||'')};
        }   
    if(M[1]==='Chrome'){
        tem=ua.match(/\bOPR\/(\d+)/)
        if(tem!=null)   {return {name:'Opera', version:tem[1]};}
        }   
    M=M[2]? [M[1], M[2]]: [navigator.appName, navigator.appVersion, '-?'];
    if((tem=ua.match(/version\/(\d+)/i))!=null) {M.splice(1,1,tem[1]);}
    return {
      name: M[0],
      version: M[1]
    };
}

function fn_nvl(value) {

}
